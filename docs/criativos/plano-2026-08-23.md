# Referências Criativos: plano de implementação

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Publicar um site estático onde clínicas navegam por pastas de criativos de referência e veem, em cada criativo, o vídeo e o roteiro pronto pra gravar.

**Architecture:** Site estático sem framework e sem dependências. `criativos.json` é a fonte de dados única. A lógica pura (roteamento por query string, busca de dados, formatação de roteiro, geração de HTML) vive em módulos ES testáveis com `node:test`; `app.js` só liga esses módulos ao DOM de cada página. Três páginas HTML: home, categoria, criativo.

**Tech Stack:** HTML + CSS + JavaScript (ES modules, sem build, sem npm install). Testes com `node --test` (runner nativo do Node 18+). Deploy no Vercel como site estático. `ffmpeg` só no script auxiliar de capas.

## Global Constraints

- Zero dependências npm. Nada de framework, nada de bundler, nada de `node_modules`.
- Todo texto de interface em português do Brasil.
- Nenhum texto do site cita preço, valor, parcelamento ou desconto.
- Tema claro travado: o CSS aplica `color-scheme: light !important` no `:root` e define cor de fundo e de texto explicitamente. Nada de bloco `prefers-color-scheme: dark`.
- Todo `<video>` de listagem usa `preload="metadata"`.
- Toda string vinda de `criativos.json` passa por escape antes de virar HTML.
- Nenhuma tela branca: todo erro e todo estado vazio tem mensagem visível.
- Nomes de arquivos, funções, variáveis e commits em português, sem acento (`acharCategoria`, `formatarRoteiroTexto`).

```js
export function lerParametros(queryString) {
  const params = new URLSearchParams(queryString);
  const limpar = (valor) => {
    if (valor === null) return null;
    const limpo = valor.trim();
    return limpo === '' ? null : limpo;
  };
  return {
    categoria: limpar(params.get('c')),
    id: limpar(params.get('id')),
  };
}
```

```js
const ERRO_CARGA = 'Não consegui carregar as referências. Recarrega a página.';

export async function carregarDados(buscar = fetch) {
  let dados;
  try {
    const resposta = await buscar('criativos.json');
    if (!resposta.ok) throw new Error(ERRO_CARGA);
    dados = await resposta.json();
  } catch {
    throw new Error(ERRO_CARGA);
  }
  if (!dados || !Array.isArray(dados.categorias)) throw new Error(ERRO_CARGA);
  return dados;
}

export function acharCategoria(dados, slug) {
  return dados.categorias.find((categoria) => categoria.slug === slug) ?? null;
}

export function acharCriativo(dados, slug, id) {
  const categoria = acharCategoria(dados, slug);
  if (!categoria) return null;
  return categoria.criativos.find((criativo) => criativo.id === id) ?? null;
}
```

```js
export function formatarRoteiroTexto(criativo) {
  const { roteiro } = criativo;
  const linhas = [criativo.titulo.toUpperCase(), '', 'GANCHO', roteiro.gancho];

  if (roteiro.desenvolvimento.length > 0) {
    linhas.push('', 'DESENVOLVIMENTO');
    roteiro.desenvolvimento.forEach((passo, indice) => {
      linhas.push(`${indice + 1}. ${passo}`);
    });
  }

  linhas.push('', 'CTA', roteiro.cta);
  linhas.push(
    '',
    'FICHA TÉCNICA',
    `Duração: ${roteiro.ficha.duracao}`,
    `Enquadramento: ${roteiro.ficha.enquadramento}`,
    `Cenário: ${roteiro.ficha.cenario}`,
    `Precisa de paciente: ${roteiro.ficha.precisa_paciente ? 'sim' : 'não'}`,
  );

  return linhas.join('\n');
}
```

```js
export function escapar(texto) {
  return String(texto)
    .replaceAll('&', '&amp;')
    .replaceAll('<', '&lt;')
    .replaceAll('>', '&gt;')
    .replaceAll('"', '&quot;')
    .replaceAll("'", '&#39;');
}

function contagem(categoria) {
  const total = categoria.criativos.length;
  if (total === 0) return 'Nenhum criativo ainda';
  return total === 1 ? '1 criativo' : `${total} criativos`;
}

export function htmlCategorias(dados) {
  return dados.categorias
    .map(
      (categoria) => `
      <a class="cartao-pasta" href="categoria.html?c=${escapar(categoria.slug)}">
        <h2>${escapar(categoria.nome)}</h2>
        <p>${escapar(categoria.descricao)}</p>
        <span class="contagem">${escapar(contagem(categoria))}</span>
      </a>`,
    )
    .join('');
}

function capaDoCartao(criativo) {
  if (criativo.capa) {
    return `<img src="${escapar(criativo.capa)}" alt="" loading="lazy">`;
  }
  if (criativo.tipo === 'arquivo') {
    return `<video src="${escapar(criativo.src)}" preload="metadata" muted playsinline></video>`;
  }
  return `<span class="capa-vazia">${escapar(criativo.titulo)}</span>`;
}

export function htmlGradeCriativos(categoria) {
  if (categoria.criativos.length === 0) {
    return htmlAviso('Ainda não tem criativo aqui. Em breve.');
  }
  return categoria.criativos
    .map(
      (criativo) => `
      <a class="cartao-criativo" href="criativo.html?c=${escapar(categoria.slug)}&amp;id=${escapar(criativo.id)}">
        <div class="moldura">${capaDoCartao(criativo)}</div>
        <h2>${escapar(criativo.titulo)}</h2>
        ${criativo.tipo === 'link' ? '<span class="etiqueta">Post externo</span>' : ''}
      </a>`,
    )
    .join('');
}

export function htmlPlayer(criativo) {
  if (criativo.tipo === 'arquivo') {
    const poster = criativo.capa ? ` poster="${escapar(criativo.capa)}"` : '';
    return `<video class="player" src="${escapar(criativo.src)}" controls playsinline preload="metadata"${poster}></video>`;
  }
  const capa = criativo.capa
    ? `<img src="${escapar(criativo.capa)}" alt="">`
    : `<span class="capa-vazia">${escapar(criativo.titulo)}</span>`;
  return `
    <div class="player player-externo">
      ${capa}
      <a class="botao" href="${escapar(criativo.src)}" target="_blank" rel="noopener noreferrer">Ver o post original</a>
    </div>`;
}

export function htmlRoteiro(criativo) {
  const { roteiro } = criativo;
  const passos = roteiro.desenvolvimento
    .map((passo) => `<li>${escapar(passo)}</li>`)
    .join('');
  const desenvolvimento =
    roteiro.desenvolvimento.length > 0
      ? `<section><h3>Desenvolvimento</h3><ol>${passos}</ol></section>`
      : '';
  return `
    <section><h3>Gancho</h3><p>${escapar(roteiro.gancho)}</p></section>
    ${desenvolvimento}
    <section><h3>CTA</h3><p>${escapar(roteiro.cta)}</p></section>
    <section class="ficha">
      <h3>Ficha técnica</h3>
      <dl>
        <dt>Duração</dt><dd>${escapar(roteiro.ficha.duracao)}</dd>
        <dt>Enquadramento</dt><dd>${escapar(roteiro.ficha.enquadramento)}</dd>
        <dt>Cenário</dt><dd>${escapar(roteiro.ficha.cenario)}</dd>
        <dt>Precisa de paciente</dt><dd>${roteiro.ficha.precisa_paciente ? 'sim' : 'não'}</dd>
      </dl>
    </section>`;
}

export function htmlAviso(mensagem) {
  return `<p class="aviso">${escapar(mensagem)}</p>`;
}
```

```js
function paginaCategoria(dados, params) {
  const categoria = acharCategoria(dados, params.categoria);
  if (!categoria) {
    window.location.replace('index.html');
    return;

function paginaCriativo(dados, params) {

  if (!criativo) {
    const destino = acharCategoria(dados, params.categoria)
      ? `categoria.html?c=${encodeURIComponent(params.categoria)}`
      : 'index.html';
```

```js
import { readdir, readFile } from 'node:fs/promises';
import { fileURLToPath } from 'node:url';
import path from 'node:path';

const raiz = path.dirname(path.dirname(fileURLToPath(import.meta.url)));

const dados = JSON.parse(await readFile(path.join(raiz, 'criativos.json'), 'utf8'));
const slugsNoJson = new Set(dados.categorias.map((categoria) => categoria.slug));
const fontesNoJson = new Set(
  dados.categorias.flatMap((categoria) => categoria.criativos.map((criativo) => criativo.src)),
);

const pastas = (await readdir(path.join(raiz, 'videos'), { withFileTypes: true }))
  .filter((entrada) => entrada.isDirectory())
  .map((entrada) => entrada.name);

const novos = [];
const fontesEmDisco = new Set();

for (const pasta of pastas) {
  if (!slugsNoJson.has(pasta)) {
    console.log(`[categoria nova] videos/${pasta} ainda não existe no criativos.json`);
  }
  const arquivos = await readdir(path.join(raiz, 'videos', pasta));
  for (const arquivo of arquivos) {
    if (!arquivo.toLowerCase().endsWith('.mp4')) continue;
    const src = `videos/${pasta}/${arquivo}`;
    fontesEmDisco.add(src);
    if (!fontesNoJson.has(src)) novos.push(src);
  }
}

for (const src of fontesNoJson) {
  if (src.startsWith('videos/') && !fontesEmDisco.has(src)) {
    console.log(`[arquivo sumiu] ${src} está no criativos.json mas não está na pasta`);
  }
}

if (novos.length === 0) {
  console.log('Nenhum vídeo novo. Tudo que está na pasta já está no site.');
} else {
  console.log(`\n${novos.length} vídeo(s) sem entrada no criativos.json:`);
  novos.forEach((src) => console.log(`  ${src}`));
}
```
